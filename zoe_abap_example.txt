*&      Form  F_LOAD_COMBO
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM f_load_combo .

  data:
        lk_comb_tab   type t_class_combo,
        lk_class_comb type t_combo.

  FIELD-SYMBOLS <fs_class_comb> like LINE OF i_class_comb.



  REFRESH i_class_comb.
  refresh i_comb_tab.

  SELECT
      kdgrp
      zz_dod_geog_desig
      zz_oc_cl_combo
      zz_oc_for_addr
      abgru
      zz_oc_combo_exist
    INTO TABLE i_class_comb
    FROM zot_class_combo.

   loop at i_class_comb into lk_class_comb.
     lk_comb_tab-zz_dod_geog_desig = lk_class_comb-zz_dod_geog_desig.
     lk_comb_tab-zz_oc_cl_combo    = lk_class_comb-zz_oc_cl_combo.
     lk_comb_tab-zz_oc_for_addr    = lk_class_comb-zz_oc_for_addr.
     lk_comb_tab-abgru             = lk_class_comb-abgru.
     lk_comb_tab-zz_oc_combo_exist  = lk_class_comb-zz_oc_combo_exist.
    append lk_comb_tab to i_comb_tab.

   ENDLOOP.


ENDFORM. " F_LOAD_COMBO
